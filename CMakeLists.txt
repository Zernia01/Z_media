cmake_minimum_required(VERSION 3.24)
project(ZerniaMedia VERSION 0.1.4 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ZM requires a 64-bit toolchain")
endif()
file(GLOB FF_ROOT "${CMAKE_SOURCE_DIR}/third_party/ffmpeg/ffmpeg-*")
file(GLOB SDL_ROOT "${CMAKE_SOURCE_DIR}/third_party/sdl/SDL2-*")
file(GLOB TTF_ROOT "${CMAKE_SOURCE_DIR}/third_party/ttf/SDL2_ttf-*")
if(WIN32)
  find_path(FF_INCLUDE libavformat/avformat.h PATHS ${FF_ROOT} PATH_SUFFIXES include REQUIRED)
  foreach(lib avformat avcodec avutil swscale swresample avfilter)
    find_library(FF_${lib} NAMES ${lib} PATHS ${FF_ROOT} PATH_SUFFIXES lib REQUIRED)
    list(APPEND MEDIA_LIBS "${FF_${lib}}")
  endforeach()
  find_path(SDL_INCLUDE SDL.h PATHS ${SDL_ROOT} PATH_SUFFIXES include REQUIRED)
  find_library(SDL_LIB SDL2 PATHS ${SDL_ROOT} PATH_SUFFIXES lib/x64 REQUIRED)
  find_path(TTF_INCLUDE SDL_ttf.h PATHS ${TTF_ROOT} PATH_SUFFIXES include REQUIRED)
  find_library(TTF_LIB SDL2_ttf PATHS ${TTF_ROOT} PATH_SUFFIXES lib/x64 REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf libavformat libavcodec libavutil libswscale libswresample libavfilter)
  list(APPEND MEDIA_LIBS PkgConfig::DEPS)
endif()
add_library(zm_core STATIC src/player/player.c src/media/decoder.c src/core/config.c src/core/utf8.c src/i18n/i18n.c src/playlist/playlist.c src/playlist/folder.c src/subtitle/subtitle.c src/input/shortcuts.c src/video/hwaccel.c)
target_include_directories(zm_core PUBLIC include)
target_include_directories(zm_core SYSTEM PUBLIC ${FF_INCLUDE} ${SDL_INCLUDE} ${TTF_INCLUDE})
target_link_libraries(zm_core PUBLIC ${MEDIA_LIBS} ${SDL_LIB} ${TTF_LIB})
target_compile_definitions(zm_core PUBLIC SDL_MAIN_HANDLED ZM_VERSION="${PROJECT_VERSION}")
add_executable(ZM WIN32 src/main.c src/ui/main_window.c src/ui/settings_window.c src/ui/controls.c src/ui/draw.c src/platform/files.c src/platform/associations.c src/platform/window.c src/platform/update.c)
if(WIN32)
  target_sources(ZM PRIVATE resources/windows.rc)
endif()
if(MSVC)
  target_sources(ZM PRIVATE resources/zm.manifest)
endif()
target_link_libraries(ZM PRIVATE zm_core)
if(WIN32)
  target_link_libraries(ZM PRIVATE comdlg32 shell32 advapi32 comctl32 winhttp bcrypt)
endif()
foreach(target ZM zm_core)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(WIN32)
  file(GLOB RUNTIME_DLLS ${FF_ROOT}/bin/*.dll ${SDL_ROOT}/lib/x64/*.dll ${TTF_ROOT}/lib/x64/*.dll)
  add_custom_command(TARGET ZM POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${RUNTIME_DLLS} $<TARGET_FILE_DIR:ZM>)
endif()
add_custom_target(zm_resources ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/resources/i18n $<TARGET_FILE_DIR:ZM>/i18n
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/resources/fonts $<TARGET_FILE_DIR:ZM>/fonts
  DEPENDS ZM)
include(CTest)
if(BUILD_TESTING)
  add_executable(zm_tests tests/core_test.c)
  target_link_libraries(zm_tests PRIVATE zm_core)
  if(MSVC)
    target_compile_options(zm_tests PRIVATE /W4 /utf-8)
  endif()
  add_test(NAME core COMMAND zm_tests)
  set(ZM_TEST_NAMES core)
  if(WIN32)
    add_executable(zm_font_tests tests/font_test.c src/ui/draw.c src/platform/files.c)
    target_link_libraries(zm_font_tests PRIVATE zm_core comdlg32 shell32)
    if(MSVC)
      target_compile_options(zm_font_tests PRIVATE /W4 /utf-8)
      target_compile_definitions(zm_font_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME mixed_script_fonts COMMAND zm_font_tests)
    list(APPEND ZM_TEST_NAMES mixed_script_fonts)
  endif()
  if(WIN32)
    add_executable(zm_update_tests tests/update_test.c src/platform/update.c)
    target_link_libraries(zm_update_tests PRIVATE zm_core winhttp bcrypt shell32)
    target_compile_definitions(zm_update_tests PRIVATE ZM_UPDATE_TESTING)
    if(MSVC)
      target_compile_options(zm_update_tests PRIVATE /W4 /utf-8)
      target_compile_definitions(zm_update_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME updater COMMAND zm_update_tests)
    list(APPEND ZM_TEST_NAMES updater)
  endif()
  add_executable(zm_smoothness_tests tests/smoothness_test.c)
  target_link_libraries(zm_smoothness_tests PRIVATE zm_core)
  if(MSVC)
    target_compile_options(zm_smoothness_tests PRIVATE /W4 /utf-8)
  endif()
  if(EXISTS "${CMAKE_SOURCE_DIR}/test-media/smoothness.mp4")
    add_test(NAME playback_smoothness COMMAND zm_smoothness_tests test-media/smoothness.mp4)
    list(APPEND ZM_TEST_NAMES playback_smoothness)
  endif()
  add_executable(zm_seek_tests tests/seek_test.c)
  target_link_libraries(zm_seek_tests PRIVATE zm_core)
  if(MSVC)
    target_compile_options(zm_seek_tests PRIVATE /W4 /utf-8)
  endif()
  if(EXISTS "${CMAKE_SOURCE_DIR}/test-media/long-seek.mkv")
    add_test(NAME long_seek COMMAND zm_seek_tests test-media/long-seek.mkv)
    list(APPEND ZM_TEST_NAMES long_seek)
  endif()
  if(WIN32)
    add_executable(zm_association_tests tests/association_test.c src/platform/associations.c)
    target_link_libraries(zm_association_tests PRIVATE zm_core shell32 advapi32)
    if(MSVC)
      target_compile_options(zm_association_tests PRIVATE /W4 /utf-8)
      target_compile_definitions(zm_association_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME associations COMMAND zm_association_tests)
    list(APPEND ZM_TEST_NAMES associations)
  endif()
  foreach(fixture sample.mp4 multiaudio.mkv)
    if(EXISTS "${CMAKE_SOURCE_DIR}/test-media/${fixture}")
      add_test(NAME playback_${fixture} COMMAND zm_tests test-media/${fixture})
      list(APPEND ZM_TEST_NAMES playback_${fixture})
    endif()
  endforeach()
  if(EXISTS "${CMAKE_SOURCE_DIR}/test-media/sample.mp4")
    add_test(NAME seek_display COMMAND ZM --smoke 1 test-media/sample.mp4)
    list(APPEND ZM_TEST_NAMES seek_display)
  endif()
  set_tests_properties(${ZM_TEST_NAMES} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "PATH=$<TARGET_FILE_DIR:ZM>\;$ENV{PATH}")
  if(TEST seek_display)
    set_property(TEST seek_display APPEND PROPERTY ENVIRONMENT "ZM_SMOKE_SEEK_DISPLAY=1")
  endif()
endif()
